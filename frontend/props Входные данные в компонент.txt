import { ref, type Ref } from 'vue';

type Callback<T extends CallbackGeneric> = (
  ...args: Parameters<T>
) => any;
type CallbackGeneric = (...args: any[]) => any;

interface UseSingltoneAsyncFunctionReturnType<T extends CallbackGeneric> {
  callback: Callback<T>;
  isLoading: Ref<boolean>;
}

export function useSingltoneAsyncFunction<T extends CallbackGeneric>(
  callback: T,
): UseSingltoneAsyncFunctionReturnType<T> {
  const isLoading = ref(false);

  let promise: Promise<ReturnType<T>> | undefined = undefined;

  return {
    callback: async (...args: Parameters<T>) => {
      isLoading.value = true;
      promise ??= callback(...args).then((res: ReturnType<T>) => {
        promise = undefined;
        isLoading.value = false;
        return res;
      });
      return <Promise<ReturnType<T>>>promise;
    },
    isLoading,
  };
}

const { callback } = UseSingltoneAsyncFunction(async (text: string | number) => {
  await async.wait(100);

  console.log(text);
}); 

callback(123);

callback('ti pidor');
